def write(path, value):

def number(t):return float(t)
def rms(x):return x.float().square().mean().sqrt()
def scatter(x):
 x=x.float();return rms(x-x.mean(-2,keepdim=True))
def paircos(x):
 x=F.normalize(x.float(),dim=-1);n=x.shape[-2]
 return ((x.sum(-2).square().sum(-1)-n)/(n*(n-1))).mean() if n>1 else x.new_tensor(1.)
def geometry(x):
 x=x.detach().float();s=scatter(x);n=rms(x)
 return {'rms':number(n),'slot_scatter_rms':number(s),'scatter_to_rms':number(s/n.clamp_min(1e-12)), 'paircos':number(paircos(x))}
def codes_geometry(x):
 x=x.reshape(38,16,256).float();g=geometry(x)
 sv=torch.linalg.svdvals(x)
 g['rank_top1_energy']=number((sv[:,0].square()/sv.square().sum(-1).clamp_min(1e-20)).mean())
 g['all_slot_paircos']=number(paircos(x.flatten(0,1)))
 g['target_mean_scatter']=number(scatter(x.mean(1)))
 return g

class Capture:
 def __init__(self,writer):self.writer=writer;self.handles=[];self.summary={};self.arrays={};self.compiler_inputs=None
 def simple(self,name,output):
  self.summary[name]=codes_geometry(output) if output.numel()==38*16*256 else geometry(output)
 def __enter__(self):
  def queryhook(m,args,out):self.arrays['language_route']=out.detach().float().cpu().numpy()
  self.handles.append(self.writer.query_language.register_forward_hook(queryhook))
  def inputs(m,args):
   self.compiler_inputs=tuple(x.detach() if torch.is_tensor(x) else x for x in args)
   self.arrays['P4']=args[1].detach().half().cpu().numpy()
   self.summary['P4']=geometry(args[1])
  self.handles.append(self.writer.compiler[0].register_forward_pre_hook(inputs))
  for i,block in enumerate(self.writer.compiler):
   def crosshook(m,args,out,i=i):
    # Same inputs/projections as real SDPA; FP32 softmax describes its logits,
    # while the actual retained output remains the untouched native SDPA result.
    q=m.split_heads(m.query(args[0]));k,v=m.project_memory(args[1],args[2])
    with torch.autocast("cuda",enabled=False):
     logits=q.float()@k.float().transpose(-2,-1)/math.sqrt(m.head_width)
     if len(args)>3 and args[3] is not None:logits=logits+args[3].float()
     p=logits.softmax(-1);n=p.shape[-1];avg=p.mean(-2,keepdim=True)
     h=-(p*p.clamp_min(1e-30).log()).sum(-1)
     delta=p-avg
     # This decomposes slot-varying attention from the shared read on actual V.
     read=<EMAIL>();shared=<EMAIL>()
     self.summary[f'cross{i}']={
       'q':geometry(q),'k':geometry(k),'v':geometry(v),'actual_output':codes_geometry(out),
       'entropy_normalized':number(h.mean()/math.log(n)) if n>1 else 0.,
       'max_attention_mean':number(p.max(-1).values.mean()),
       'attention_slot_rms':number(rms(delta)),
       'attention_slot_l1':number(delta.abs().sum(-1).mean()),
       'attention_slot_relative_rms':number(rms(delta)/rms(p).clamp_min(1e-12)),
       'pre_output_read':geometry(read),
       'read_slot_relative_rms':number(rms(read-shared)/rms(read).clamp_min(1e-12)),
       'logit_time_std':number(logits.std(-1).mean()),
       'first_quarter_mass':number(p[...,:max(1,n//4)].sum(-1).mean()),
       'last_quarter_mass':number(p[...,-max(1,n//4):].sum(-1).mean())}
     self.arrays[f'cross{i}_attention_mean']=p.mean(-2).cpu().numpy()
     self.arrays[f'cross{i}_entropy']=h.cpu().numpy()
   self.handles.append(block.cross.register_forward_hook(crosshook))
   def blockhook(m,args,out,i=i):self.simple(f'block{i}_output',out)
   self.handles.append(block.register_forward_hook(blockhook))
   for key,mod in [('self',block.self_attention),('ffn',block.ffn)]:
    def part(m,args,out,name=f'block{i}_{key}'):self.simple(name,out)
    self.handles.append(mod.register_forward_hook(part))
  for i,group in enumerate(self.writer.process_groups):
   def grouphook(m,args,out,i=i):
    states,process=out
    self.summary[f'group{i}']={'horizon':geometry(states),'process':geometry(process),'horizon_frame_mean':geometry(states.mean(1))}
   self.handles.append(group.register_forward_hook(grouphook))
  def decoderhook(m,args,out):
   c=args[0];a=F.gelu(m.a_code(c));b=F.gelu(m.b_code(c))
   self.summary['C']=codes_geometry(c);self.summary['H_A']=codes_geometry(a);self.summary['H_B']=codes_geometry(b)
   self.arrays['C']=c.detach().half().cpu().numpy()
   self.arrays['H_A_task_mean']=a.mean(1).float().cpu().numpy()
   self.arrays['H_B_task_mean']=b.mean(1).float().cpu().numpy()
   # Small Gram matrices supply gauge-independent BA energy and rank separation.
   with torch.autocast("cuda",enabled=False):
    vals=[]
    for target in m.contract.targets:
     aa=out[target.name+LORA_A_SUFFIX].float();bb=out[target.name+LORA_B_SUFFIX].float()
     ra=torch.linalg.qr(aa.T,mode='r').R;rb=torch.linalg.qr(bb,mode='r').R
     ss=torch.linalg.svdvals(rb@ra.T)
     vals.append({'B_rank_paircos':number(paircos(bb.T)), 'BA_norm':number(ss.norm()),
                  'BA_top1_energy':number(ss[0].square()/ss.square().sum().clamp_min(1e-20))})
   self.summary['outputs_by_target']=vals
  self.handles.append(self.writer.decoder.register_forward_hook(decoderhook))
  return self
 def __exit__(self,*_):
  for h in self.handles:h.remove()


def main():

  mc=dict(record['model_config']);mode=mc.pop('backend_conditioning','all');mc['decoder_rank_sharing']='within_target' if arm=='shared7' else 'independent'
  assert mode==('local_h_read' if arm.startswith('off') else 'all')
  if runtime.state.writer.config.decoder_rank_sharing!=mc['decoder_rank_sharing']:
   runtime.state.writer=HorizonRelationWriter(runtime.lora,HorizonWriterConfig(**mc)).to(device)
  for step in [200,400]:

   runtime.state.requires_grad_(False).eval();writer=runtime.state.writer
   handle=writer.query_language.register_forward_hook(lambda m,a,o:torch.zeros_like(o)) if arm.startswith('off') else None
   for c in conditions:
    if time.monotonic()-start>args.max_seconds or sum(p.stat().st_size for p in O.iterdir() if p.is_file())>2**30:
     status.update(state='budget_stop',seconds=time.monotonic()-start);write(O/'runtime.json',status);return
    key=f'{arm}_{step}_task{c["task"]:02d}_video{c["ordinal"]}';tick=time.monotonic()
