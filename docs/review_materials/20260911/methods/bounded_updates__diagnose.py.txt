"""Four temporary real macro updates and zero-current-gradient Adam controls."""
import argparse,copy,importlib.util,json,os,time
from pathlib import Path
import torch
from safetensors.torch import load_file,save_file

ROOT=Path('<REPO_ROOT>');HERE=Path(__file__).resolve().parent
RUN=ROOT/'runs/outputs/horizon_k1_frame_contextual_v1_seed7_20260909'
CKPT=RUN/'checkpoints/macro_00000400'
VICTIMS={0:(33,48),19:(32,48),22:(32,47),34:(33,49)}
ARMS=('normal','M','P','C','D','PC','PD','CD','PCD')
KIND='bounded_four_update_cross_task_mechanism'
def read(p):return json.loads(Path(p).read_text())
def write(p,v):
    q=p.with_suffix('.tmp');q.write_text(json.dumps(v,indent=2,allow_nan=False)+'\n');q.replace(p)
def group(name):
    if name.startswith('writer.decoder.'):return 'D'
    if name.startswith(('writer.target_queries','writer.rank_queries','writer.query_language.','writer.time_projection.','writer.compiler.')):return 'C'
    return 'P'
def panel_module():
    spec=importlib.util.spec_from_file_location('existing_video_panel',HERE.parent/'video_controls/panel.py')
    m=importlib.util.module_from_spec(spec);spec.loader.exec_module(m);m.KIND=KIND;return m
def snapshot(state):return {k:v.detach().cpu().clone() for k,v in state.state_dict().items()}

def prepare():
    from ember.pi05_eval_queue import EvaluationShard,initialize_queue
    from ember.writer.materialization import source_matches
    m=panel_module();manifest=read(HERE/'manifest.json');original=read(m.SOURCE)
    assert manifest['status']=='complete' and manifest['model_updates']==4
    assert source_matches(manifest['source'],original['adapter']['source'])
    tasks=sorted((t for t in original['tasks'] if m.gid(t) in VICTIMS),key=m.gid)
    assert tuple(m.gid(t) for t in tasks)==tuple(VICTIMS)
    for t in tasks:
        assert t['split_role']=='train';t['init_state_ids']=[VICTIMS[m.gid(t)][0]]
    out=HERE/'closedloop';out.mkdir();(out/'shards').mkdir();(out/'workers').mkdir()
    (out/'libero_config').symlink_to(m.SOURCE.parent/'libero_config',target_is_directory=True)
    contract={k:original[k] for k in ('model','normalization','tokenizer','libero_paths','policy','environment','rng')}
    contract.update(schema_version='ember_bounded_update_diagnostic_v1',role='development_train',mode='diagnostic',kind=KIND,
        tasks=tasks,parallel={'envs_per_replica':1},adapter=None,diagnostic_exploration=None,diagnostic_occupancy_capture=None,
        diagnostic_stage_predicates={'schema_version':'ember_pi05_stage_predicate_capture_v1',
        'capture':'post_settling_then_every_executed_action_change_points','predicate_source':'installed_LIBERO_BDDL_goal_conjunction',
        'training_gradient_use':False,'checkpoint_selection_use':False},reference_contract=str(m.SOURCE),
        diagnostic_manifest=str(HERE/'manifest.json'),writer_checkpoint=manifest['writer_checkpoint'],
        arms=list(ARMS),camera_view='agentview',status='prepared',expected_rows=36,jobs={})
    shards=[];lookup={m.gid(t):t for t in tasks}
    for c in manifest['conditions']:
        t=lookup[c['global_task_id']];name=c['condition_id'];assert c['language']==t['language']
        contract['jobs'][name]={k:c[k] for k in ('global_task_id','arm','adapter_path','source_task','teacher_demo','camera_view','condition_id')}
        shards.append(EvaluationShard(name,len(shards),t['suite'],t['task_id'],t['horizon'],tuple(t['init_state_ids']),t['horizon']))
    write(out/'contract.json',contract);initialize_queue(out/'queue.sqlite3',shards,contract_reference='ember_four_update_diagnostic_20260910')

def materialize(args):
    from ember.pi05_source_setup import initialize_distributed
    from ember.pi05_source_checkpoint import restore_rng
    from ember.writer.runtime import build_runtime,FrozenVideoPrefixCache
    from ember.writer.learning_data import WriterTrainingData
    from ember.writer.training import _optimization,_update
    from ember.writer.supervised import SupervisedEngine
    from ember.writer.native import autocast
    from ember.writer.materialization import source_matches
    from ember.lora import identity_lora_state,copy_task_lora_state_,validate_lora_state
    reg=read(HERE/'registration.json');assert reg['status']=='registered_before_results'
    assert not (HERE/'manifest.json').exists();(HERE/'adapters').mkdir()
    torch.set_num_threads(4);torch.manual_seed(7);torch.backends.cuda.matmul.allow_tf32=True
    ctx=initialize_distributed(require_numa=True,defer_process_group=True);device=ctx.device
    uuid='GPU-'+str(torch.cuda.get_device_properties(device).uuid).removeprefix('GPU-')
    assert ctx.world_size==1 and torch.cuda.device_count()==1 and uuid==os.environ['EXPECTED_GPU_UUID']
    run=read(RUN/'run_contract.json');cfg=copy.deepcopy(run['config'])
    assert cfg['observer'].get('camera_view','agentview')=='agentview'
    cfg['runtime']['policy_microbatch']=4;cfg['runtime']['prefix_cache_bytes']=2*2**30
    runtime=build_runtime(ROOT,cfg,device);assert source_matches(runtime.source,run['source'])
    original=load_file(str(CKPT/'ecp.safetensors'),device='cpu');runtime.state.load_state_dict(original,strict=True)
    trainer=torch.load(CKPT/'trainer_state.pt',map_location='cpu',weights_only=False,mmap=True)
    rank=torch.load(CKPT/'rank_00_state.pt',map_location='cpu',weights_only=False,mmap=True)
    assert trainer['sampler_state']['next_step']==400 and trainer['scheduler']['last_epoch']==400
    data=WriterTrainingData(ROOT,cfg['data']);data.restore_sampler(trainer['sampler_state'])
    # Preview and restore only the temporary sampler. No batch/victim search.
    draws=[data.next_iteration() for _ in range(4)];data.restore_sampler(trainer['sampler_state'])
    assert [[x['task'] for x in ds] for ds in draws]==reg['next_macro_tasks']
    assert not set(VICTIMS)&{x['task'] for ds in draws for x in ds}
    optimizer,scheduler=_optimization(runtime.state,cfg)
    optimizer.load_state_dict(copy.deepcopy(trainer['optimizer']));scheduler.load_state_dict(copy.deepcopy(trainer['scheduler']))
    restore_rng(rank['rng'],ctx);runtime.state.requires_grad_(True).train();runtime.policy.requires_grad_(False)
    cache=FrozenVideoPrefixCache(runtime.observer,data,cfg['runtime']['prefix_cache_bytes'])
    engine=SupervisedEngine(runtime,data,cache,ctx,cfg)
    tick=time.monotonic();torch.cuda.reset_peak_memory_stats(device)
    manifest={'status':'running','kind':KIND,'camera_view':'agentview','source':runtime.source,'writer_checkpoint':{
        'source':str(CKPT),'temporary_update_count':4,'hybrids_not_formal_checkpoints':True},'model_updates':0,
        'arms':list(ARMS),'task_ids':list(VICTIMS),'gpu_uuid':uuid,'conditions':[],
        'topology':'single GPU serial equal-weight four-condition diagnostic; not world4 exact resume',
        'actual_updates':[],'current_gradient_zero_updates':[],'original_outputs_modified':False}
    write(HERE/'manifest.json',manifest)
    try:
        for step in range(401,405):
            if time.monotonic()-tick>args.max_seconds:raise TimeoutError('four-update wall budget')
            rows,norms=_update(engine,runtime,data,ctx,cfg,optimizer,scheduler,step)
            assert len(rows)==4 and not set(VICTIMS)&{r['task'] for r in rows}
            assert all(not p.requires_grad and p.grad is None for p in runtime.policy.parameters())
            manifest['actual_updates'].append({'step':step,'conditions':rows,**norms})
            manifest['model_updates']+=1;write(HERE/'manifest.json',manifest)
            print(json.dumps({'update':step,'seconds':time.monotonic()-tick,**norms}),flush=True)
        full=snapshot(runtime.state);optimizer.zero_grad(set_to_none=True)
        runtime.state.load_state_dict(original,strict=True)
        optimizer.load_state_dict(copy.deepcopy(trainer['optimizer']));scheduler.load_state_dict(copy.deepcopy(trainer['scheduler']))
        for step in range(401,405):
            for p in runtime.state.parameters():p.grad=torch.zeros_like(p)
            optimizer.step();scheduler.step()
            manifest['current_gradient_zero_updates'].append({'step':step,'current_gradient':0,'lr_next':scheduler.get_last_lr()[0]})
        momentum=snapshot(runtime.state);optimizer.zero_grad(set_to_none=True)
        del optimizer,scheduler,engine,cache,trainer,rank
        runtime.state.requires_grad_(False).eval();runtime.policy.requires_grad_(False).eval()
        copy_task_lora_state_(runtime.policy,identity_lora_state(runtime.lora),runtime.lora)
        manifest['sampler_after']=data.sampler_state()['next_step'];assert manifest['sampler_after']==404
        with torch.no_grad():
            for task,(_,demo) in VICTIMS.items():
                videos,indices=data.load_videos(task,(demo,))
                condition=runtime.observer.prepare(videos,indices,data.tasks[task].authority.language)
                inputs=runtime.observer.writer_arguments(condition)
                for arm in ARMS:
                    if time.monotonic()-tick>args.max_seconds:raise TimeoutError('four-update materialization budget')
                    state=original if arm=='normal' else {k:(full[k] if group(k) in arm else momentum[k]) for k in original}
                    runtime.state.load_state_dict(state,strict=True)
                    with autocast(device):
                        responses=runtime.observer.responses(condition);generated=runtime.state.writer(responses,*inputs)
                    validate_lora_state(generated,runtime.lora);assert all(torch.isfinite(v).all() for v in generated.values())
                    name=f'task{task:02d}_{arm}';path=HERE/'adapters'/f'{name}.safetensors'
                    save_file({k:v.detach().cpu().contiguous() for k,v in generated.items()},str(path))
                    manifest['conditions'].append({'condition_id':name,'global_task_id':task,'source_task':task,'arm':arm,
                        'language':data.tasks[task].authority.language,'teacher_demo':demo,'camera_view':'agentview',
                        'adapter_path':str(path),'complete_target_count':38,'tensor_count':76})
                    write(HERE/'manifest.json',manifest);del responses,generated
                del condition,inputs,videos
        assert len(manifest['conditions'])==36
        manifest.update(status='complete',seconds=time.monotonic()-tick,peak_gib=torch.cuda.max_memory_allocated(device)/2**30)
        write(HERE/'manifest.json',manifest);prepare()
    except Exception as error:
        manifest.update(status='failed',error=repr(error));write(HERE/'manifest.json',manifest);raise
    finally:data.close()

if __name__=='__main__':
    ap=argparse.ArgumentParser();ap.add_argument('mode',choices=('materialize','worker'));ap.add_argument('--worker');ap.add_argument('--gpu',type=int);ap.add_argument('--max-seconds',type=int,default=1800);args=ap.parse_args()
    if args.mode=='materialize':materialize(args)
    else:panel_module().worker(HERE,args)
